cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(codex_over_cloudflare VERSION 0.5.20 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(WIN32 AND (NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_GENERATOR_PLATFORM MATCHES "^(ARM|Win32)" OR CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM|arm|aarch)"))
 message(FATAL_ERROR "Windows requires an AMD64 toolchain (x64)")
endif()
if(MSVC)
 if(VCPKG_TARGET_TRIPLET MATCHES "-static$")
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 endif()
 add_compile_options(/W4 /permissive- /utf-8 /bigobj)
else()
 add_compile_options(-Wall -Wextra -Wpedantic)
endif()
if(WIN32)
 add_compile_definitions(_WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
option(COC_STATIC_MUSL "Build fully static Linux executables with a musl toolchain" OFF)
if(COC_STATIC_MUSL)
 if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "COC_STATIC_MUSL requires a Linux musl toolchain")
 endif()
 include(CheckCXXSourceCompiles)
 check_cxx_source_compiles("#include <features.h>
 #ifdef __GLIBC__
 #error COC_STATIC_MUSL requires musl, not glibc
 #endif
 int main() { return 0; }" COC_NON_GLIBC_TOOLCHAIN)
 if(NOT COC_NON_GLIBC_TOOLCHAIN)
  message(FATAL_ERROR "COC_STATIC_MUSL requires a musl toolchain; use Dockerfile.musl")
 endif()
 set(Boost_USE_STATIC_LIBS ON)
 set(OPENSSL_USE_STATIC_LIBS TRUE)
 set(CMAKE_FIND_LIBRARY_SUFFIXES .a)
 add_link_options(-static)
 add_compile_definitions(COC_STATIC_MUSL=1)
endif()
find_package(Boost 1.71 REQUIRED CONFIG COMPONENTS system)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(nlohmann_json 3.7 REQUIRED)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_target(coc_build_identity ALL
 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-metadata.py ${CMAKE_CURRENT_BINARY_DIR}/generated/coc/build.hpp
 WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
 VERBATIM
 BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/generated/coc/build.hpp)
option(COC_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(COC_SANITIZE AND MSVC)
 message(FATAL_ERROR "COC_SANITIZE requires GCC or Clang sanitizers")
endif()
if(COC_SANITIZE AND COC_STATIC_MUSL)
 message(FATAL_ERROR "COC_SANITIZE cannot be combined with fully static musl builds")
endif()
if(COC_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
if(WIN32)
 set(COC_PLATFORM_SOURCES native/src/windows/platform.cpp native/src/windows/files.cpp)
else()
 set(COC_PLATFORM_SOURCES native/src/files.cpp)
endif()
add_library(coc_core ${COC_PLATFORM_SOURCES} native/src/projects.cpp native/src/update.cpp native/src/external.cpp native/src/approvals.cpp native/src/socket_unix.cpp native/src/login.cpp native/src/common.cpp native/src/store.cpp native/src/delivery.cpp native/src/delivery_client.cpp native/src/process.cpp native/src/network.cpp native/src/access.cpp native/src/proxy.cpp native/src/tui_bridge.cpp native/src/socket_plain.cpp native/src/socket_tls.cpp)
add_custom_target(coc_access_defaults ALL
 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-access.py ${CMAKE_CURRENT_BINARY_DIR}/generated/coc/access_defaults.hpp
 WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
 VERBATIM
 BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/generated/coc/access_defaults.hpp)
add_dependencies(coc_core coc_build_identity coc_access_defaults)
target_include_directories(coc_core PUBLIC native/include ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(coc_core PUBLIC Boost::system OpenSSL::SSL OpenSSL::Crypto SQLite::SQLite3 nlohmann_json::nlohmann_json Threads::Threads)
if(WIN32)
 target_link_libraries(coc_core PUBLIC ws2_32 mswsock crypt32 advapi32)
endif()
add_executable(coc-agent native/src/agent.cpp)
target_link_libraries(coc-agent PRIVATE coc_core)
add_executable(coc-client native/src/cli.cpp)
target_link_libraries(coc-client PRIVATE coc_core)
include(CTest)
if(BUILD_TESTING)
 if(WIN32)
  add_executable(coc-tests native/tests/windows.cpp)
  add_executable(coc-test-child native/tests/windows-child.cpp)
  target_link_libraries(coc-test-child PRIVATE coc_core)
 else()
  add_executable(coc-tests native/tests/core.cpp)
 endif()
 target_link_libraries(coc-tests PRIVATE coc_core)
 target_compile_definitions(coc-tests PRIVATE COC_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 add_executable(coc-project-tests native/tests/projects.cpp)
 target_link_libraries(coc-project-tests PRIVATE coc_core)
 add_test(NAME native-projects COMMAND coc-project-tests)
 set_tests_properties(native-projects PROPERTIES TIMEOUT 60)
 add_test(NAME native-core COMMAND coc-tests)
 set_tests_properties(native-core PROPERTIES TIMEOUT 300)
endif()
install(TARGETS coc-client coc-agent RUNTIME DESTINATION bin)
option(COC_FUZZ "Build LLVM libFuzzer protocol harness (Clang required)" OFF)
if(COC_FUZZ)
 add_executable(coc-protocol-fuzz native/tests/fuzz.cpp)
 target_link_libraries(coc-protocol-fuzz PRIVATE coc_core)
 target_compile_options(coc-protocol-fuzz PRIVATE -fsanitize=fuzzer)
 target_link_options(coc-protocol-fuzz PRIVATE -fsanitize=fuzzer)
endif()
