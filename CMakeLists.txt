cmake_minimum_required(VERSION 3.16)
project(codex_over_cloudflare VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Boost 1.71 REQUIRED CONFIG COMPONENTS system)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(nlohmann_json 3.7 REQUIRED)
find_package(Threads REQUIRED)
add_library(coc_core native/src/login.cpp native/src/common.cpp native/src/store.cpp native/src/process.cpp native/src/network.cpp native/src/socket_plain.cpp native/src/socket_tls.cpp)
target_include_directories(coc_core PUBLIC native/include)
target_link_libraries(coc_core PUBLIC Boost::system OpenSSL::SSL OpenSSL::Crypto SQLite::SQLite3 nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(coc_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(coc-agent native/src/agent.cpp)
target_link_libraries(coc-agent PRIVATE coc_core)
target_compile_options(coc-agent PRIVATE -Wall -Wextra -Wpedantic)
add_executable(coc-client native/src/cli.cpp)
target_link_libraries(coc-client PRIVATE coc_core)
target_compile_options(coc-client PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
 add_executable(coc-tests native/tests/core.cpp)
 target_link_libraries(coc-tests PRIVATE coc_core)
 target_compile_definitions(coc-tests PRIVATE COC_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 add_test(NAME native-core COMMAND coc-tests)
endif()
install(TARGETS coc-client coc-agent RUNTIME DESTINATION bin)
