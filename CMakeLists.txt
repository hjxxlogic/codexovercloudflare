cmake_minimum_required(VERSION 3.16)
project(codex_over_cloudflare VERSION 0.5.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Boost 1.71 REQUIRED CONFIG COMPONENTS system)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(nlohmann_json 3.7 REQUIRED)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_target(coc_build_identity ALL
 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-metadata.py ${CMAKE_CURRENT_BINARY_DIR}/generated/coc/build.hpp
 WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
 BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/generated/coc/build.hpp)
option(COC_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(COC_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(coc_core native/src/update.cpp native/src/files.cpp native/src/login.cpp native/src/common.cpp native/src/store.cpp native/src/process.cpp native/src/network.cpp native/src/socket_plain.cpp native/src/socket_tls.cpp)
add_dependencies(coc_core coc_build_identity)
target_include_directories(coc_core PUBLIC native/include ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(coc_core PUBLIC Boost::system OpenSSL::SSL OpenSSL::Crypto SQLite::SQLite3 nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(coc_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(coc-agent native/src/agent.cpp)
target_link_libraries(coc-agent PRIVATE coc_core)
target_compile_options(coc-agent PRIVATE -Wall -Wextra -Wpedantic)
add_executable(coc-client native/src/cli.cpp)
target_link_libraries(coc-client PRIVATE coc_core)
target_compile_options(coc-client PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
 add_executable(coc-tests native/tests/core.cpp)
 target_link_libraries(coc-tests PRIVATE coc_core)
 target_compile_definitions(coc-tests PRIVATE COC_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 add_test(NAME native-core COMMAND coc-tests)
 set_tests_properties(native-core PROPERTIES TIMEOUT 300)
endif()
install(TARGETS coc-client coc-agent RUNTIME DESTINATION bin)
option(COC_FUZZ "Build LLVM libFuzzer protocol harness (Clang required)" OFF)
if(COC_FUZZ)
 add_executable(coc-protocol-fuzz native/tests/fuzz.cpp)
 target_link_libraries(coc-protocol-fuzz PRIVATE coc_core)
 target_compile_options(coc-protocol-fuzz PRIVATE -fsanitize=fuzzer)
 target_link_options(coc-protocol-fuzz PRIVATE -fsanitize=fuzzer)
endif()
